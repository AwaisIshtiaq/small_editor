   129→function Field({
   130→  label,
   131→  hint,
   132→  children,
   133→}: {
   134→  label: string
   135→  hint?: string
   136→  children: React.ReactNode
   137→}) {
   138→  return (
   139→    <div className="space-y-1.5">
   140→      <Label className="text-sm">{label}</Label>
   141→      {children}
   142→      {hint ? <p className="text-xs text-muted-foreground">{hint}</p> : null}
   143→    </div>
   144→  )
   145→}

   147→function TextField({
   148→  label,
   149→  value,
   150→  onChange,
   151→  placeholder,
   152→  hint,
   153→  type = 'text',
   154→}: {
   155→  label: string
   156→  value: string
   157→  onChange: (v: string) => void
   158→  placeholder?: string
   159→  hint?: string
   160→  type?: string
   161→}) {
   162→  return (
   163→    <Field label={label} hint={hint}>
   164→      <Input
   165→        type={type}
   166→        value={value}
   167→        placeholder={placeholder}
   168→        onChange={(e) => onChange(e.target.value)}
   169→      />
   170→    </Field>
   171→  )
   172→}

   174→function NumField({
   175→  label,
   176→  value,
   177→  onChange,
   178→  min,
   179→  max,
   180→  step,
   181→}: {
   182→  label: string
   183→  value: number
   184→  onChange: (v: number) => void
   185→  min?: number
   186→  max?: number
   187→  step?: number
   188→}) {
   189→  return (
   190→    <Field label={label}>
   191→      <Input
   192→        type="number"
   193→        value={Number.isFinite(value) ? String(value) : ''}
   194→        min={min}
   195→        max={max}
   196→        step={step}
   197→        onChange={(e) => {
   198→          const n = e.target.valueAsNumber
   199→          onChange(Number.isFinite(n) ? n : 0)
   200→        }}
   201→      />
   202→    </Field>
   203→  )
   204→}

   206→function SelectField({
   207→  label,
   208→  value,
   209→  onChange,
   210→  options,
   211→  hint,
   212→}: {
   213→  label: string
   214→  value: string
   215→  onChange: (v: string) => void
   216→  options: { value: string; label: string }[]
   217→  hint?: string
   218→}) {
   219→  return (
   220→    <Field label={label} hint={hint}>
   221→      <Select value={value} onValueChange={onChange}>
   222→        <SelectTrigger className="w-full">
   223→          <SelectValue />
   224→        </SelectTrigger>
   225→        <SelectContent>
   226→          {options.map((o) => (
   227→            <SelectItem key={o.value} value={o.value}>
   228→              {o.label}
   229→            </SelectItem>
   230→          ))}
   231→        </SelectContent>
   232→      </Select>
   233→    </Field>
   234→  )
   235→}

   237→function BoolField({
   238→  label,
   239→  checked,
   240→  onChange,
   241→}: {
   242→  label: string
   243→  checked: boolean
   244→  onChange: (v: boolean) => void

   246→  const id = useId()
   247→  return (
   248→    <div className="flex items-center gap-2">
   249→      <Checkbox id={id} checked={checked} onCheckedChange={(v) => onChange(v === true)} />
   250→      <Label htmlFor={id} className="text-sm font-normal">
   251→        {label}
   252→      </Label>
   253→    </div>
   254→  )

  1093→// Minimal client-side required-field checks (server re-validates).
  1094→export function optionError(slug: string, params: Params): string | null {
  1095→  const v = params as Record<string, string | number | boolean | unknown[]>
  1096→  switch (slug) {
  1097→    case 'split':
  1098→      return v.mode === 'ranges' && !String(v.ranges).trim() ? 'Enter at least one page range.' : null
  1099→    case 'extract-pages':
  1100→    case 'delete-pages':
  1101→      return !String(v.pages).trim() ? 'Enter the pages to process.' : null
